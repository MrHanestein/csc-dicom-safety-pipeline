def deterministic_uid(old_uid: str, secret_salt: str) -> UID:
    clean_old = str(old_uid).strip()
    digest = hmac.new(secret_salt.encode("utf-8"), clean_old.encode("utf-8"), hashlib.sha256).digest()
    int_128 = int.from_bytes(digest[:16], byteorder="big")
    return UID(f"2.25.{int_128}")


def deterministic_pseudonym(raw_id: str, secret_salt: str, length: int = 12) -> str:
    digest = hmac.new(secret_salt.encode("utf-8"), str(raw_id).strip().encode("utf-8"), hashlib.sha256).hexdigest()
    return f"ANON-{digest[:length].upper()}"

def recursive_clean_with_walk(ds: Dataset, secret_salt: str) -> None:
    ds.remove_private_tags()

    def _clean_element(curr_ds: Dataset, elem: pydicom.DataElement) -> None:
        tag_tuple = (elem.tag.group, elem.tag.element)
        if elem.VR == "SQ":
            if tag_tuple not in PS315_ALLOWLIST_POLICY:
                del curr_ds[elem.tag]
            return
        if tag_tuple not in PS315_ALLOWLIST_POLICY:
            del curr_ds[elem.tag]
            return
        action, _ = PS315_ALLOWLIST_POLICY[tag_tuple]
        if action == "Z":
            elem.value = ""
        elif action == "D" and tag_tuple == (0x0008, 0x0070):
            elem.value = "ANON_VENDOR"
        elif action == "C":
            elem.value = deterministic_pseudonym(elem.value, secret_salt)
        elif action == "U":
            elem.value = deterministic_uid(elem.value, secret_salt)

    ds.walk(_clean_element)


def deidentify_dicom_stage1_to_4(input_path: Path, output_path: Path, secret_salt: str) -> Dataset:
    ds = pydicom.dcmread(input_path)
    recursive_clean_with_walk(ds, secret_salt)

    ds.PatientIdentityRemoved = "YES"
    ds.DeidentificationMethod = "PS3.15_Annex_E_AllowList_Profile"
    ds.file_meta.MediaStorageSOPInstanceUID = ds.SOPInstanceUID

    ds.save_as(output_path, enforce_file_format=True)
    return ds


def stage5_leak_validator_gate(
    deid_filepath: Path,
    forbidden_strings: list[str],
    allowlist_policy: dict = PS315_ALLOWLIST_POLICY
) -> tuple[bool, list[str]]:
    violations = []
    ds = pydicom.dcmread(deid_filepath)

    def _inspect_element(curr_ds: Dataset, elem: pydicom.DataElement) -> None:
        tag_tuple = (elem.tag.group, elem.tag.element)
        if elem.tag.group % 2 == 1:
            violations.append(f"HAZARD-PRIV-TAG: Private odd-group tag {tag_tuple} survived!")
        if elem.VR != "SQ":
            if tag_tuple not in allowlist_policy:
                violations.append(f"HAZARD-UNAPPROVED-TAG: Tag {tag_tuple} ({elem.name}) is not on allow-list!")
            else:
                action, keyword = allowlist_policy[tag_tuple]
                if action == "Z" and str(elem.value).strip() != "":
                    violations.append(f"HAZARD-NONEMPTY-Z: Tag {keyword} {tag_tuple} must be blank!")
            if tag_tuple != (0x7FE0, 0x0010):
                val_str = str(elem.value).upper()
                for secret in forbidden_strings:
                    if secret and len(secret) >= 3 and secret.upper() in val_str:
                        violations.append(
                            f"HAZARD-PHI-LEAK: Forbidden string '{secret}' leaked inside {elem.name} {tag_tuple}!"
                        )

    ds.walk(_inspect_element)

    if getattr(ds, "PatientIdentityRemoved", "") != "YES":
        violations.append("HAZARD-MISSING-MARKER: PatientIdentityRemoved (0012,0062) is not 'YES'!")

    return (len(violations) == 0), violations


def run_pipeline_with_audit_log(
    input_path: Path,
    output_path: Path,
    audit_log_path: Path,
    secret_salt: str = "GSTT_CSC_SALT_2026"
) -> dict:
    raw_ds = pydicom.dcmread(input_path)
    forbidden_phi = [
        str(getattr(raw_ds, "PatientName", "")),
        str(getattr(raw_ds, "PatientID", "")),
        str(getattr(raw_ds, "InstitutionName", "")),
        str(getattr(raw_ds, "ReferringPhysicianName", "")),
    ]

    with open(input_path, "rb") as f:
        input_sha256 = hashlib.file_digest(f, "sha256").hexdigest()

    deidentify_dicom_stage1_to_4(input_path, output_path, secret_salt)
    passed_gate, violations = stage5_leak_validator_gate(output_path, forbidden_phi)

    if not passed_gate:
        if output_path.exists():
            output_path.unlink()
        status = "BLOCKED_BY_STAGE5_GATE"
        output_sha256 = None
    else:
        status = "VERIFIED_SAFE"
        with open(output_path, "rb") as f:
            output_sha256 = hashlib.file_digest(f, "sha256").hexdigest()

    audit_record = {
        "timestamp_utc": datetime.now(timezone.utc).isoformat(),
        "input_file": str(input_path.name),
        "input_sha256": input_sha256,
        "output_sha256": output_sha256,
        "status": status,
        "violations_detected": violations,
    }
    with open(audit_log_path, "a", encoding="utf-8") as f:
        f.write(json.dumps(audit_record) + "\n")

    return audit_record

class StudyRegistry:
    def __init__(self, db_path: str = "study_registry.sqlite"):
        self.conn = sqlite3.connect(db_path)
        self.conn.execute("""
            CREATE TABLE IF NOT EXISTS study_registry (
                input_sha256 TEXT PRIMARY KEY,
                patient_pseudonym TEXT NOT NULL,
                deid_study_uid TEXT NOT NULL,
                status TEXT NOT NULL,
                registered_at_utc TEXT NOT NULL
            )
        """)
        self.conn.commit()

    def register_study_idempotent(self, input_sha256: str, pseudonym: str, deid_study_uid: str, status: str) -> bool:
        cursor = self.conn.execute("SELECT 1 FROM study_registry WHERE input_sha256 = ?", (input_sha256,))
        if cursor.fetchone() is not None:
            return False
        self.conn.execute(
            "INSERT INTO study_registry VALUES (?, ?, ?, ?, ?)",
            (input_sha256, pseudonym, deid_study_uid, status, datetime.now(timezone.utc).isoformat())
        )
        self.conn.commit()
        return True

    def count_studies(self) -> int:
        return self.conn.execute("SELECT COUNT(*) FROM study_registry").fetchone()[0]


def upload_verified_dicom_to_orthanc(
    deid_filepath: Path,
    forbidden_strings: list[str],
    orthanc_url: str = "http://127.0.0.1:8042/instances",
    username: str = "orthanc",
    password: str = "orthanc"
) -> dict:
    is_safe, violations = stage5_leak_validator_gate(deid_filepath, forbidden_strings)
    if not is_safe:
        raise PermissionError(f"HAZ-09 BLOCK: Refusing Orthanc upload! Violations: {violations}")

    dicom_bytes = deid_filepath.read_bytes()
    req = urllib.request.Request(orthanc_url, data=dicom_bytes, method="POST")
    req.add_header("Content-Type", "application/dicom")
    b64_auth = base64.b64encode(f"{username}:{password}".encode("utf-8")).decode("ascii")
    req.add_header("Authorization", f"Basic {b64_auth}")

    with urllib.request.urlopen(req, timeout=10) as response:
        return json.loads(response.read().decode("utf-8"))

We claim that `csc-dicom-safety-pipeline` reduces the risk of (a) unauthorized Patient Identifiable Information (PHI) leakage and (b) silent corruption of quantitative CT physics metadata (`PixelSpacing`, `SliceThickness`, `RescaleSlope`, `RescaleIntercept`) to an acceptable level (**Residual Risk Score <= 4 across all 15 identified hazards in `CSC-F-013`**).

## 2. Summary of Deterministic Engineering Controls & Test Evidence
1. **Strict Allow-List Enforcement (`PS315_ALLOWLIST_POLICY`):** Replaces error-prone block-lists by stripping all unapproved tags and all odd-group vendor private tags (`group % 2 == 1`).
2. **Stage 5 Blocking Leak Validator Gate (`stage5_leak_validator_gate`):** Re-reads the serialized output file from disk using recursive `ds.walk()` and physically deletes (`unlink()`) any file containing residual PHI, odd private tags, or missing `PatientIdentityRemoved == "YES"` markers prior to Orthanc network transmission.
3. **Automated CI Verification (`tests/test_safety_gate.py`):** Verified on every commit via GitHub Actions against deliberately planted PHI leaks (`HAZ-01` to `HAZ-15`).

## 3. What Would Be Required Before Touching Real NHS Patient Data
1. Formal **Data Protection Impact Assessment (DPIA)** and **Caldicott Guardian** approval within the NHS Trust.
2. Hardware Security Module (HSM) or Kubernetes Secret management for the HMAC-SHA256 `secret_salt` (stored on an air-gapped Trust VLAN).
3. Optical Character Recognition (OCR) pixel-scrubbing gate for modalities with burned-in pixel annotations (Ultrasound / Secondary Capture).
4. Local **DCB0160** Clinical Risk Management sign-off by the receiving NHS Trust's appointed Clinical Safety Officer (CSO).
